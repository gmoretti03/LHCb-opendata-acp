\section{Notation}
\label{sec:notation}
\begin{table}[h]
\centering
\begin{tabular}{lp{6.2cm}}
\toprule
Symbol & Meaning \\
\midrule
$\Araw$ & raw charge asymmetry $(N^--N^+)/(N^-+N^+)$ of the signal, parameter $A$ of the fit \\
$\ACP$, $A_P$, $A_D$ & CP asymmetry; production asymmetry of $B^\pm$; detection asymmetry \\
$A_{\mathrm{count}}$ & counting asymmetry of all candidates of a sample \\
$A_b$ & asymmetry of the background yields \\
$N_s$, $N_b$ & signal and background yields (both charges) \\
$\mu$, $\sigma$ & mean and width of the Gaussian signal \\
$\lambda_{1,2}^\pm$, $f_\pm$ & slopes and fraction of the double-exponential background, per charge \\
$\theta$ & vector of the twelve parameters \\
$\mathcal L$, $\pi$ & likelihood and prior \\
$q(A)=2\Delta\mathrm{NLL}$ & profile likelihood ratio of $A$; $t(A_{\mathrm{true}})$ at the injected value \\
$Z$, $p$ & significance and $p$-value of the test of $A=0$ \\
$BF_{01}$, $BF_{10}$ & Bayes factor of $A=0$ against $A$ free, and its inverse \\
$p_B$ & posterior predictive $p$-value \\
$\sigma_B$ & expected spread of the shift between nested samples, $\sqrt{|\sigma_{\mathrm{var}}^2-\sigma_{\mathrm{nom}}^2|}$ \\
$N/\tau$ & chain length in integrated autocorrelation times \\
$S_i$, $S_k$ & local significance of a Dalitz bin or cluster with respect to the global asymmetry \\
\bottomrule
\end{tabular}
\caption{Main symbols used in the text.}
\label{tab:notation}
\end{table}
